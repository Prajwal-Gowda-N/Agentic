from groq import Groq

client = Groq(
    api_key="API KEY"
)

response = client.chat.completions.create(
    model="openai/gpt-oss-20b",
    messages=[
        {
            "role":"user",
            "content":"""
Goal:
Organize a college technical fest.

Create a detailed step-by-step plan.
"""
        }
    ]
)

print(response.choices[0].message.content)
